+/**
+ * Validate that a YYYY-MM-DD string represents an actual calendar date (e.g., rejects 2026-02-30).
+ */
+export function isValidIsoCalendarDate(value: string): boolean {
+  if (!/^\d{4}-\d{2}-\d{2}$/.test(value)) {
+    return false;
+  }
+  const [yearStr, monthStr, dayStr] = value.split("-");
+  const year = Number(yearStr);
+  const month = Number(monthStr);
+  const day = Number(dayStr);
+
+  if (!Number.isInteger(year) || !Number.isInteger(month) || !Number.isInteger(day)) {
+    return false;
+  }
+  if (month < 1 || month > 12 || day < 1 || day > 31) {
+    return false;
+  }
+
+  const utc = new Date(Date.UTC(year, month - 1, day));
+  return (
+    utc.getUTCFullYear() === year &&
+    utc.getUTCMonth() === month - 1 &&
+    utc.getUTCDate() === day
+  );
+}

+function parseIbgeNumeric(raw: unknown): number | null {
+  if (typeof raw === "number") {
+    return Number.isFinite(raw) ? raw : null;
+  }
+  if (typeof raw !== "string") {
+    return null;
+  }
+  const trimmed = raw.trim();
+  if (
+    trimmed.length === 0 ||
+    trimmed === "-" ||
+    trimmed === "..." ||
+    trimmed === ".." ||
+    trimmed === "X"
+  ) {
+    return null;
+  }
+  const parsed = Number(trimmed);
+  return Number.isFinite(parsed) ? parsed : null;

+function parseIbgeAggregatedResponse(
+  rawData: unknown,
+  kind: SingleIbgeIndicator,
+  fallbackNivel: "N3" | "N6",
+  fallbackCodigo: string,
+  sourceUrl: string,
+): IbgeIndicatorSeriesResult | null {
+  if (!Array.isArray(rawData) || rawData.length === 0) {
+    return null;
+  }
+
+  const entry = rawData[0];
+  if (typeof entry !== "object" || entry === null) {
+    return null;
+  }
+
+  const meta = INDICATOR_META[kind];
+  const record = entry as Record<string, unknown>;
+  const resultados = record.resultados;
+  if (!Array.isArray(resultados) || resultados.length === 0) {
+    return null;
+  }
+
+  let firstSeries: Record<string, unknown> | null = null;
+  for (const res of resultados) {
+    if (typeof res !== "object" || res === null) {
+      continue;
+    }
+    const seriesList = (res as { series?: unknown }).series;
+    if (Array.isArray(seriesList) && seriesList.length > 0) {
+      const candidate = seriesList[0];
+      if (typeof candidate === "object" && candidate !== null) {
+        firstSeries = candidate as Record<string, unknown>;
+        break;
+      }
+    }
+  }
+
+  if (!firstSeries) {
+    return null;
+  }
+
+  const rawSerie = firstSeries.serie;
+  if (typeof rawSerie !== "object" || rawSerie === null || Array.isArray(rawSerie)) {
+    return null;
+  }
+
+  const serieEntries = Object.entries(rawSerie as Record<string, unknown>)
+    .map(([key, value]) => [sanitizeUntrustedText(key, 32), parseIbgeNumeric(value)] as const)
+    .filter(([key]) => key.length > 0)
+    .sort(([a], [b]) => a.localeCompare(b));
+
+  if (serieEntries.length === 0) {
+    return null;
+  }
+
+  const serie: Record<string, number | null> = {};
+  for (const [periodKey, value] of serieEntries) {
+    serie[periodKey] = value;
+  }
+
+  let latestPeriod: string | null = null;
+  let latestValue: number | null = null;
+  for (let i = serieEntries.length - 1; i >= 0; i--) {
+    const [periodKey, value] = serieEntries[i];
+    if (value !== null) {
+      latestPeriod = periodKey;
+      latestValue = value;
+      break;
+    }
+  }
+  if (latestPeriod === null) {
+    latestPeriod = serieEntries[serieEntries.length - 1][0];
+  }
+
+  const rawLoc =
+    typeof firstSeries.localidade === "object" && firstSeries.localidade !== null
+      ? (firstSeries.localidade as Record<string, unknown>)
+      : {};
+  const rawNivel =
+    typeof rawLoc.nivel === "object" && rawLoc.nivel !== null
+      ? (rawLoc.nivel as Record<string, unknown>)
+      : {};
+
+  return {
+    agregado_id: meta.agregadoId,
+    variavel_id: meta.variavelId,
+    variavel: sanitizeUntrustedText(
+      typeof record.variavel === "string" && record.variavel.trim().length > 0
+        ? record.variavel
+        : meta.defaultVariavel,
+      300,
+    ),
+    unidade: sanitizeUntrustedText(
+      typeof record.unidade === "string" && record.unidade.trim().length > 0
+        ? record.unidade
+        : meta.defaultUnidade,
+      100,
+    ),
+    localidade: {
+      id: sanitizeUntrustedText(
+        typeof rawLoc.id === "string" || typeof rawLoc.id === "number"
+          ? String(rawLoc.id)
+          : fallbackCodigo,
+        32,
+      ),
+      nome: sanitizeUntrustedText(
+        typeof rawLoc.nome === "string" && rawLoc.nome.trim().length > 0
+          ? rawLoc.nome
+          : fallbackCodigo,
+        300,
+      ),
+      nivel: {
+        id: sanitizeUntrustedText(
+          typeof rawNivel.id === "string" && rawNivel.id.trim().length > 0
+            ? rawNivel.id
+            : fallbackNivel,
+          32,
+        ),
+        nome: sanitizeUntrustedText(
+          typeof rawNivel.nome === "string" && rawNivel.nome.trim().length > 0
+            ? rawNivel.nome
+            : fallbackNivel === "N3"
+              ? "Unidade da Federação"
+              : "Município",
+          100,
+        ),
+      },
+    },
+    serie,
+    latest_period: latestPeriod,
+    latest_value: latestValue,
+    source_url: sourceUrl,
+  };

+function computePibPerCapita(
+  pop: IbgeIndicatorSeriesResult | null,
+  pib: IbgeIndicatorSeriesResult | null,
+): {
+  valor: number | null;
+  referencia: PibPerCapitaDetails | null;
+} {
+  if (!pop || !pib) {
+    return { valor: null, referencia: null };
+  }
+
+  const commonPeriods = Object.keys(pib.serie)
+    .filter((period) => {
+      const pibVal = pib.serie[period];
+      const popVal = pop.serie[period];
+      return (
+        pibVal !== null &&
+        pibVal !== undefined &&
+        popVal !== null &&
+        popVal !== undefined &&
+        popVal > 0
+      );
+    })
+    .sort((a, b) => b.localeCompare(a));
+
+  let pibAno = pib.latest_period;
+  let popAno = pop.latest_period;
+  let pibVal = pib.latest_value;
+  let popVal = pop.latest_value;
+
+  if (commonPeriods.length > 0) {
+    const matchedYear = commonPeriods[0];
+    pibAno = matchedYear;
+    popAno = matchedYear;
+    pibVal = pib.serie[matchedYear];
+    popVal = pop.serie[matchedYear];
+  }
+
+  if (
+    !pibAno ||
+    !popAno ||
+    pibVal === null ||
+    popVal === null ||
+    !Number.isFinite(pibVal) ||
+    !Number.isFinite(popVal) ||
+    popVal <= 0
+  ) {
+    return { valor: null, referencia: null };
+  }
+
+  const multiplier = pib.unidade.toLowerCase().includes("mil") ? 1000 : 1;
+  const valor = Math.round(((pibVal * multiplier) / popVal) * 100) / 100;
+  const mesmoAno = pibAno === popAno;
+
+  return {
+    valor,
+    referencia: {
+      pib_ano: pibAno,
+      populacao_ano: popAno,
+      mesmo_ano: mesmoAno,
+      formula:
+        multiplier === 1000
+          ? "(pib_mil_reais * 1000) / populacao"
+          : "pib_reais / populacao",
+      nota: mesmoAno
+        ? `Cálculo realizado com PIB e População do mesmo ano de referência (${pibAno}).`
+        : `PIB (${pibAno}) e População (${popAno}) pertencem a anos de referência distintos; resultado é uma razão estimada.`,
+    },
+  };
+}

+async function fetchSingleIndicator(
+  kind: SingleIbgeIndicator,
+  periodo: string,
+  nivel: "N3" | "N6",
+  codigoIbge: string,
+): Promise<{
+  url: string;
+  httpOk: boolean;
+  error?: string;
+  parsed: IbgeIndicatorSeriesResult | null;
+}> {
+  const url = buildIbgeIndicatorUrl(kind, periodo, nivel, codigoIbge);
+  const httpRes = await httpJson(url);
+  if (!httpRes.ok) {
+    return {
+      url,
+      httpOk: false,
+      error: httpRes.error || "request failed",
+      parsed: null,
+    };
+  }
+  return {
+    url,
+    httpOk: true,
+    parsed: parseIbgeAggregatedResponse(httpRes.data, kind, nivel, codigoIbge, url),
+  };
+}

+export async function executeIbgeIndicadores(
+  args: IbgeIndicadoresInput,
+  cache?: Cache,
+): Promise<LookupResult> {
+  const parsed = IbgeIndicadoresSchema.safeParse(args ?? {});
+  if (!parsed.success) {
+    const details = parsed.error.issues
+      .map((issue) => {
+        const path = issue.path.length > 0 ? `${issue.path.join(".")}: ` : "";
+        return `${path}${issue.message}`;
+      })
+      .join("; ");
+    return {
+      ok: false,
+      error: `Invalid parameters for ibge_indicadores: ${details}`,
+    };
+  }
+
+  const { codigo_ibge, indicador, periodo } = parsed.data;
+  const nivel: "N3" | "N6" = codigo_ibge.length === 2 ? "N3" : "N6";
+  const cacheKey = `ibge_indicadores:${codigo_ibge}:${indicador}:${periodo}`;
+
+  try {
+    if (cache) {
+      const cached = await cache.get<CachedPayload>(cacheKey);
+      if (cached && typeof cached === "object" && "data" in cached && cached.source) {
+        return { ok: true, data: cached.data, source: cached.source };
+      }
+    }
+
+    if (indicador === "populacao" || indicador === "pib") {
+      const single = await fetchSingleIndicator(indicador, periodo, nivel, codigo_ibge);
+      if (!single.httpOk) {
+        return {
+          ok: false,
+          error: `IBGE Agregados API error: ${single.error}`,
+        };
+      }
+      if (!single.parsed) {
+        return {
+          ok: false,
+          error: `IBGE indicators not found for codigo_ibge: ${codigo_ibge}`,
+        };
+      }
+
+      const fetchedAt = new Date().toISOString();
+      const payload = {
+        codigo_ibge,
+        nivel,
+        indicador,
+        periodo,
+        localidade: single.parsed.localidade,
+        unidade: single.parsed.unidade,
+        serie: single.parsed.serie,
+        latest_period: single.parsed.latest_period,
+        latest_value: single.parsed.latest_value,
+        [indicador]: single.parsed,
+        source: "IBGE — Agregados v3",
+        source_url: single.url,
+        fetchedAt,
+        untrusted_data_notice: UNTRUSTED_EXTERNAL_DATA_NOTICE,
+      };
+      const source = { url: single.url, fetchedAt };
+      if (cache) {
+        await cache.set(cacheKey, { data: payload, source });
+      }
+      return { ok: true, data: payload, source };
+    }
+
+    // indicador === "todos" -> fetch population and GDP in parallel
+    const [popOutcome, pibOutcome] = await Promise.all([
+      fetchSingleIndicator("populacao", periodo, nivel, codigo_ibge),
+      fetchSingleIndicator("pib", periodo, nivel, codigo_ibge),
+    ]);
+
+    if (!popOutcome.httpOk && !pibOutcome.httpOk) {
+      return {
+        ok: false,
+        error: `IBGE Agregados API error: ${popOutcome.error || pibOutcome.error}`,
+      };
+    }
+
+    const warnings: string[] = [];
+    if (!popOutcome.httpOk) {
+      warnings.push(`${INDICATOR_META.populacao.label} lookup failed: ${popOutcome.error}`);
+    }
+    if (!pibOutcome.httpOk) {
+      warnings.push(`${INDICATOR_META.pib.label} lookup failed: ${pibOutcome.error}`);
+    }
+
+    const pop = popOutcome.parsed;
+    const pib = pibOutcome.parsed;
+
+    if (!pop && !pib) {
+      if (warnings.length > 0) {
+        return {
+          ok: false,
+          error: `IBGE Agregados API error: ${warnings.join("; ")}`,
+        };
+      }
+      return {
+        ok: false,
+        error: `IBGE indicators not found for codigo_ibge: ${codigo_ibge}`,
+      };
+    }
+
+    const localidade = (pop ?? pib)!.localidade;
+    const perCapita = computePibPerCapita(pop, pib);
+    const fetchedAt = new Date().toISOString();
+    const payload = {
+      codigo_ibge,
+      nivel,
+      indicador,
+      periodo,
+      localidade,
+      populacao: pop,
+      pib,
+      pib_per_capita_estimado_reais: perCapita.valor,
+      pib_per_capita_referencia: perCapita.referencia,
+      ...(warnings.length > 0 ? { warnings } : {}),
+      source: "IBGE — Agregados v3",
+      source_url: popOutcome.url,
+      source_urls: [popOutcome.url, pibOutcome.url],
+      fetchedAt,
+      untrusted_data_notice: UNTRUSTED_EXTERNAL_DATA_NOTICE,
+    };
+
+    const source = { url: popOutcome.url, fetchedAt };
+    if (cache && warnings.length === 0) {
+      await cache.set(cacheKey, { data: payload, source });
+    }
+    return { ok: true, data: payload, source };
+  } catch (err) {
+    return {
+      ok: false,
+      error: `IBGE Agregados error: ${err instanceof Error ? err.message : String(err)}`,
+    };
+  }
+}

+function buildPtaxUrl(

+function buildFocusUrl(
+  indicador: FocusIndicator,
+  anoReferencia?: number,
+): string {
+  // Filter terms are guaranteed by strict Zod enums and bounded integer validation.
+  const filterExpr =
+    anoReferencia !== undefined
+      ? `Indicador eq '${indicador}' and DataReferencia eq '${anoReferencia}'`
+      : `Indicador eq '${indicador}'`;
+  // Olinda's Apache Olingo servlet requires %20 rather than + for spaces and literal $ keys.
+  const encodedFilter = new URLSearchParams([["f", filterExpr]])
+    .toString()
+    .slice(2)
+    .replace(/\+/g, "%20");
+  return `${OLINDA_FOCUS_URL}?$filter=${encodedFilter}&$top=10&$orderby=Data%20desc&$format=json`;

+function normalizePtaxQuotes(
+  rawData: unknown,
+  targetDate: string,
+): NormalizedPtaxQuote[] {
+  if (typeof rawData !== "object" || rawData === null) {
+    return [];
+  }
+  const value = (rawData as { value?: unknown }).value;
+  if (!Array.isArray(value)) {
+    return [];
+  }
+
+  const normalized: NormalizedPtaxQuote[] = [];
+  for (const entry of value) {
+    if (typeof entry !== "object" || entry === null) {
+      continue;
+    }
+    const record = entry as Record<string, unknown>;
+    const cotacaoCompra = toFiniteNumberOrNull(record.cotacaoCompra);
+    const cotacaoVenda = toFiniteNumberOrNull(record.cotacaoVenda);
+    const rawDateTime =
+      typeof record.dataHoraCotacao === "string"
+        ? sanitizeUntrustedText(record.dataHoraCotacao, 64)
+        : "";
+    const dateMatch = /^(\d{4}-\d{2}-\d{2})/.exec(rawDateTime);
+
+    if (
+      cotacaoCompra === null ||
+      cotacaoVenda === null ||
+      !dateMatch ||
+      !isValidIsoCalendarDate(dateMatch[1]) ||
+      dateMatch[1] > targetDate
+    ) {
+      continue;
+    }
+
+    const tipoBoletim =
+      typeof record.tipoBoletim === "string" && record.tipoBoletim.trim().length > 0
+        ? sanitizeUntrustedText(record.tipoBoletim, 100)
+        : null;
+    const paridadeCompra = toFiniteNumberOrNull(record.paridadeCompra);
+    const paridadeVenda = toFiniteNumberOrNull(record.paridadeVenda);
+
+    normalized.push({
+      effectiveDate: dateMatch[1],
+      cotacaoCompra,
+      cotacaoVenda,
+      dataHoraCotacao: rawDateTime,
+      tipoBoletim,
+      ...(paridadeCompra !== null ? { paridadeCompra } : {}),
+      ...(paridadeVenda !== null ? { paridadeVenda } : {}),
+    });
+  }
+
+  normalized.sort((a, b) => {
+    if (a.effectiveDate !== b.effectiveDate) {
+      return b.effectiveDate.localeCompare(a.effectiveDate);
+    }
+    return b.dataHoraCotacao.localeCompare(a.dataHoraCotacao);
+  });
+
+  return normalized;
+}

+function selectPreferredQuote(quotes: NormalizedPtaxQuote[]): NormalizedPtaxQuote {
+  const closingQuotes = quotes.filter(
+    (q) => q.tipoBoletim === "Fechamento PTAX" || q.tipoBoletim === "Fechamento",
+  );
+  if (closingQuotes.length > 0) {
+    const topClosingDate = closingQuotes[0].effectiveDate;
+    const onTopDate = closingQuotes.filter(
+      (q) => q.effectiveDate === topClosingDate,
+    );
+    return (
+      onTopDate.find((q) => q.tipoBoletim === "Fechamento PTAX") ??
+      onTopDate.find((q) => q.tipoBoletim === "Fechamento") ??
+      onTopDate[0]
+    );
+  }
+  return quotes[0];
+}

+async function readFromCache(
+  cache: Cache | undefined,
+  cacheKey: string,
+): Promise<LookupResult | null> {
+  if (!cache) {
+    return null;
+  }
+  const cached = await cache.get<CachedPayload>(cacheKey);
+  if (cached && typeof cached === "object" && "data" in cached && cached.source) {
+    return { ok: true, data: cached.data, source: cached.source };
+  }
+  return null;

+async function executePtax(
+  moeda: OlindaCurrency,
+  targetDate: string,
+  cache?: Cache,
+): Promise<LookupResult> {
+  const cacheKey = `olinda:ptax:${moeda}:${targetDate}`;
+  const cached = await readFromCache(cache, cacheKey);
+  if (cached) {
+    return cached;
+  }
+
+  const startIso = shiftIsoDate(targetDate, -7);
+  const url = buildPtaxUrl(moeda, toOlindaDate(startIso), toOlindaDate(targetDate));
+  const result = await httpJson(url);
+
+  if (!result.ok) {
+    return {
+      ok: false,
+      error: `Banco Central Olinda PTAX API error: ${result.error || "request failed"}`,
+    };
+  }
+
+  const quotes = normalizePtaxQuotes(result.data, targetDate);
+  if (quotes.length === 0) {
+    return {
+      ok: false,
+      error: `No PTAX quote found for ${moeda} up to ${targetDate} (checked 7-day window starting ${startIso}).`,
+    };
+  }
+
+  const selected = selectPreferredQuote(quotes);
+  const fetchedAt = new Date().toISOString();
+  const payload = {
+    consulta: "ptax" as const,
+    moeda,
+    requested_date: targetDate,
+    effective_date: selected.effectiveDate,
+    exact_date_match: selected.effectiveDate === targetDate,
+    cotacao_compra: selected.cotacaoCompra,
+    cotacao_venda: selected.cotacaoVenda,
+    data_hora_cotacao: selected.dataHoraCotacao,
+    tipo_boletim: selected.tipoBoletim,
+    ...(selected.paridadeCompra !== undefined
+      ? { paridade_compra: selected.paridadeCompra }
+      : {}),
+    ...(selected.paridadeVenda !== undefined
+      ? { paridade_venda: selected.paridadeVenda }
+      : {}),
+    recent_quotes: quotes.slice(0, 5).map((q) => ({
+      effective_date: q.effectiveDate,
+      cotacao_compra: q.cotacaoCompra,
+      cotacao_venda: q.cotacaoVenda,
+      data_hora_cotacao: q.dataHoraCotacao,
+      tipo_boletim: q.tipoBoletim,
+      ...(q.paridadeCompra !== undefined
+        ? { paridade_compra: q.paridadeCompra }
+        : {}),
+      ...(q.paridadeVenda !== undefined
+        ? { paridade_venda: q.paridadeVenda }
+        : {}),
+    })),
+    source: "Banco Central do Brasil — Olinda PTAX",
+    source_url: url,
+    fetchedAt,
+    untrusted_data_notice: UNTRUSTED_EXTERNAL_DATA_NOTICE,
+  };
+
+  const source = { url, fetchedAt };
+  if (cache) {
+    await cache.set(cacheKey, { data: payload, source });
+  }
+  return { ok: true, data: payload, source };
+}

+async function executeFocus(
+  indicador: FocusIndicator,
+  anoReferencia: number | undefined,
+  cache?: Cache,
+): Promise<LookupResult> {
+  const cacheKey = `olinda:focus:${indicador}:${anoReferencia ?? "all"}`;
+  const cached = await readFromCache(cache, cacheKey);
+  if (cached) {
+    return cached;
+  }
+
+  const url = buildFocusUrl(indicador, anoReferencia);
+  const result = await httpJson(url);
+
+  if (!result.ok) {
+    return {
+      ok: false,
+      error: `Banco Central Olinda Focus API error: ${result.error || "request failed"}`,
+    };
+  }
+
+  if (
+    typeof result.data !== "object" ||
+    result.data === null ||
+    !Array.isArray((result.data as { value?: unknown }).value)
+  ) {
+    return {
+      ok: false,
+      error: "Unexpected response format from Banco Central Olinda Focus API",
+    };
+  }
+
+  const rawList = (result.data as { value: unknown[] }).value;
+  const expectations = rawList
+    .filter(
+      (entry): entry is Record<string, unknown> =>
+        typeof entry === "object" && entry !== null,
+    )
+    .map((entry) => ({
+      indicador: sanitizeUntrustedText(
+        typeof entry.Indicador === "string" ? entry.Indicador : indicador,
+        100,
+      ),
+      indicador_detalhe:
+        typeof entry.IndicadorDetalhe === "string" &&
+        entry.IndicadorDetalhe.trim().length > 0
+          ? sanitizeUntrustedText(entry.IndicadorDetalhe, 200)
+          : null,
+      data_pesquisa:
+        typeof entry.Data === "string"
+          ? sanitizeUntrustedText(entry.Data, 40)
+          : null,
+      ano_referencia:
+        typeof entry.DataReferencia === "string" ||
+        typeof entry.DataReferencia === "number"
+          ? sanitizeUntrustedText(String(entry.DataReferencia), 20)
+          : null,
+      media: toFiniteNumberOrNull(entry.Media),
+      mediana: toFiniteNumberOrNull(entry.Mediana),
+      minimo: toFiniteNumberOrNull(entry.Minimo),
+      maximo: toFiniteNumberOrNull(entry.Maximo),
+      desvio_padrao: toFiniteNumberOrNull(entry.DesvioPadrao),
+      respondentes: toFiniteNumberOrNull(entry.numeroRespondentes),
+    }));
+
+  const fetchedAt = new Date().toISOString();
+  const payload = {
+    consulta: "focus" as const,
+    indicador,
+    ...(anoReferencia !== undefined ? { ano_referencia: anoReferencia } : {}),
+    total: expectations.length,
+    results: expectations,
+    source: "Banco Central do Brasil — Olinda Expectativas Focus",
+    source_url: url,
+    fetchedAt,
+    untrusted_data_notice: UNTRUSTED_EXTERNAL_DATA_NOTICE,
+  };
+
+  const source = { url, fetchedAt };
+  if (cache) {
+    await cache.set(cacheKey, { data: payload, source });
+  }
+  return { ok: true, data: payload, source };
+}

+export async function executeBcbOlinda(
+  args: BcbOlindaInput,
+  cache?: Cache,
+): Promise<LookupResult> {
+  const parsed = BcbOlindaSchema.safeParse(args ?? {});
+  if (!parsed.success) {
+    const details = parsed.error.issues
+      .map((issue) => {
+        const path = issue.path.length > 0 ? `${issue.path.join(".")}: ` : "";
+        return `${path}${issue.message}`;
+      })
+      .join("; ");
+    return {
+      ok: false,
+      error: `Invalid parameters for bcb_olinda: ${details}`,
+    };
+  }
+
+  const { consulta, data, moeda, indicador, ano_referencia } = parsed.data;
+  try {
+    if (consulta === "ptax") {
+      const targetDate = data ?? todaySaoPauloIso();
+      if (!isValidIsoCalendarDate(targetDate)) {
+        return {
+          ok: false,
+          error: `Invalid date for bcb_olinda: ${targetDate}`,
+        };
+      }
+      return await executePtax(moeda, targetDate, cache);
+    }
+
+    return await executeFocus(indicador, ano_referencia, cache);
+  } catch (err) {
+    return {
+      ok: false,
+      error: `Banco Central Olinda error: ${err instanceof Error ? err.message : String(err)}`,
+    };
+  }
+}
