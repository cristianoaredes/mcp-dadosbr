+/**
+ * Decode MTE Lista Suja CSV bytes as UTF-8, falling back to ISO-8859-1 on invalid UTF-8 sequences.
+ */
+export function decodeMteCsv(buffer: ArrayBuffer): string {
+  try {
+    return new TextDecoder("utf-8", { fatal: true }).decode(buffer);
+  } catch {
+    return new TextDecoder("iso-8859-1").decode(buffer);
+  }

+function normalizeHeader(value: string): string {
+  return value
+    .trim()
+    .normalize("NFD")
+    .replace(/[\u0300-\u036f]/g, "")
+    .toLowerCase();
+}

+function normalizeEmployerForSearch(value: string): string {
+  return value
+    .normalize("NFD")
+    .replace(/[\u0300-\u036f]/g, "")

+function parseSemicolonCsvRows(csvText: string): string[][] {
+  const input = csvText.charCodeAt(0) === 0xfeff ? csvText.slice(1) : csvText;
+  const rows: string[][] = [];
+  let currentRow: string[] = [];
+  let currentField = "";
+  let inQuotes = false;
+
+  for (let i = 0; i < input.length; i++) {
+    const ch = input[i];
+
+    if (inQuotes) {
+      if (ch === '"') {
+        if (input[i + 1] === '"') {
+          currentField += '"';
+          i++;
+        } else {
+          inQuotes = false;
+        }
+      } else {
+        currentField += ch;
+      }
+      continue;
+    }
+
+    if (ch === '"' && currentField.length === 0) {
+      inQuotes = true;
+    } else if (ch === ";") {
+      currentRow.push(currentField);
+      currentField = "";
+    } else if (ch === "\r") {
+      if (input[i + 1] === "\n") {
+        i++;
+      }
+      currentRow.push(currentField);
+      rows.push(currentRow);
+      currentRow = [];
+      currentField = "";
+    } else if (ch === "\n") {
+      currentRow.push(currentField);
+      rows.push(currentRow);
+      currentRow = [];
+      currentField = "";
+    } else {
+      currentField += ch;
+    }
+  }
+
+  if (currentField.length > 0 || currentRow.length > 0) {
+    currentRow.push(currentField);
+    rows.push(currentRow);
+  }
+
+  return rows.filter((row) => row.some((cell) => cell.trim() !== ""));
+}

+function findColumnIndex(
+  normalizedHeaders: string[],
+  exactNames: string[],
+  partialCheck?: (header: string) => boolean,
+): number {
+  const exactIdx = normalizedHeaders.findIndex((h) => exactNames.includes(h));
+  if (exactIdx !== -1) {
+    return exactIdx;
+  }
+  if (partialCheck) {
+    return normalizedHeaders.findIndex(partialCheck);
+  }
+  return -1;

+function resolveColumns(headerCells: string[]): ColumnMap {
+  if (
+    headerCells.length < 2 ||
+    headerCells.some((cell) => /<[a-z!/]/i.test(cell.trim()))
+  ) {
+    throw new Error("Invalid MTE Lista Suja CSV header");
+  }
+
+  const normalized = headerCells.map(normalizeHeader);
+
+  const empregador = findColumnIndex(
+    normalized,
+    ["empregador", "nome empregador", "razao social"],
+    (h) => h.includes("empregador") && !h.includes("inclusao"),

+  const resolvedEmpregador =
+    empregador !== -1
+      ? empregador
+      : normalized.findIndex((h) => h.includes("empregador"));
+
+  const cnpj_cpf = findColumnIndex(
+    normalized,
+    ["cnpj/cpf", "cpf/cnpj", "cnpj", "cpf"],
+    (h) => h.includes("cnpj"),
+  );
+
+  if (
+    resolvedEmpregador === -1 ||
+    cnpj_cpf === -1 ||
+    resolvedEmpregador === cnpj_cpf
+  ) {
+    throw new Error("Invalid MTE Lista Suja CSV header");
+  }
+
+  return {
+    id: findColumnIndex(normalized, ["id"]),
+    ano_acao_fiscal: findColumnIndex(
+      normalized,
+      ["ano da acao fiscal"],
+      (h) => h.includes("ano") && h.includes("acao"),
+    ),
+    uf: findColumnIndex(normalized, ["uf"]),
+    empregador: resolvedEmpregador,
+    cnpj_cpf,
+    estabelecimento: findColumnIndex(
+      normalized,
+      ["estabelecimento"],
+      (h) => h.includes("estabelecimento"),
+    ),
+    trabalhadores_envolvidos: findColumnIndex(
+      normalized,
+      ["trabalhadores envolvidos"],
+      (h) => h.includes("trabalhadores"),
+    ),
+    cnae: findColumnIndex(normalized, ["cnae"], (h) => h.includes("cnae")),
+    decisao_procedencia: findColumnIndex(
+      normalized,
+      ["decisao administrativa de procedencia", "decisao de procedencia"],
+      (h) => h.includes("decisao") || h.includes("procedencia"),
+    ),
+    inclusao_cadastro: findColumnIndex(
+      normalized,
+      ["inclusao no cadastro de empregadores", "inclusao no cadastro"],
+      (h) => h.includes("inclusao"),
+    ),
+  };
+}

+function readCell(row: string[], index: number): string {
+  if (index < 0 || index >= row.length) {
+    return "";
+  }
+  return sanitizeUntrustedText(row[index] ?? "", 300);
+}

+/**
+ * Parse MTE Lista Suja semicolon-delimited CSV text into sanitized structured entries.
+ */
+export function parseMteListaSujaCsv(csvText: string): MteListaSujaEntry[] {
+  const rows = parseSemicolonCsvRows(csvText);
+  if (rows.length === 0) {
+    throw new Error("Invalid MTE Lista Suja CSV header");
+  }
+
+  const columns = resolveColumns(rows[0]);
+  const entries: MteListaSujaEntry[] = [];
+
+  for (let i = 1; i < rows.length; i++) {
+    const row = rows[i];
+    const empregador = readCell(row, columns.empregador);
+    const cnpj_cpf = readCell(row, columns.cnpj_cpf);
+    if (!empregador && !cnpj_cpf) {
+      continue;
+    }
+
+    entries.push({
+      id: readCell(row, columns.id),
+      ano_acao_fiscal: readCell(row, columns.ano_acao_fiscal),
+      uf: readCell(row, columns.uf),
+      empregador,
+      cnpj_cpf,
+      cnpj_cpf_digits: cnpj_cpf.replace(/\D/g, ""),
+      estabelecimento: readCell(row, columns.estabelecimento),
+      trabalhadores_envolvidos: readCell(row, columns.trabalhadores_envolvidos),
+      cnae: readCell(row, columns.cnae),
+      decisao_procedencia: readCell(row, columns.decisao_procedencia),
+      inclusao_cadastro: readCell(row, columns.inclusao_cadastro),
+      normalized_employer: normalizeEmployerForSearch(empregador),
+    });
+  }
+
+  return entries;
+}

+function toPublicEntry(entry: MteListaSujaEntry): MteListaSujaPublicEntry {
+  const { cnpj_cpf_digits: _digits, normalized_employer: _norm, ...publicFields } = entry;
+  return publicFields;
+}

+/**
+ * Filter parsed MTE Lista Suja entries by CNPJ/CPF digits (>= 8 digits) or accent-insensitive employer tokens.
+ */
+export function matchMteListaSuja(
+  entries: MteListaSujaEntry[],
+  query: string,
+  maxMatches: number = 50,
+): MteListaSujaMatchResult {
+  const limit = Number.isFinite(maxMatches) && maxMatches > 0 ? Math.floor(maxMatches) : 0;
+  const digits = query.replace(/\D/g, "");
+
+  let matched: MteListaSujaEntry[];
+  if (digits.length >= 8) {
+    const exactLength = digits.length === 11 || digits.length === 14;
+    matched = entries.filter((entry) =>
+      exactLength
+        ? entry.cnpj_cpf_digits === digits
+        : entry.cnpj_cpf_digits.includes(digits),
+    );
+  } else {
+    const tokens = normalizeEmployerForSearch(query)
+      .split(/\s+/)
+      .filter(Boolean);
+    if (tokens.length === 0) {
+      return { total_matches: 0, matches: [] };
+    }
+    matched = entries.filter((entry) =>
+      tokens.every((token) => entry.normalized_employer.includes(token)),
+    );
+  }
+
+  return {
+    total_matches: matched.length,
+    matches: matched.slice(0, limit).map(toPublicEntry),
+  };
+}

+/**
+ * Validate 14-digit CNPJ checksum using mod-11 algorithm.
+ */
+export function validateTcuCnpjChecksum(cnpj: string): boolean {
+  if (cnpj.length !== 14 || !/^\d{14}$/.test(cnpj)) {
+    return false;
+  }
+
+  if (/^(\d)\1{13}$/.test(cnpj)) {
+    return false;
+  }
+
+  let sum = 0;
+  let multiplier = 5;
+  for (let i = 0; i < 12; i++) {
+    sum += parseInt(cnpj[i], 10) * multiplier;
+    multiplier = multiplier === 2 ? 9 : multiplier - 1;
+  }
+  const remainder1 = sum % 11;
+  const digit1 = remainder1 < 2 ? 0 : 11 - remainder1;
+
+  sum = 0;
+  multiplier = 6;
+  for (let i = 0; i < 13; i++) {
+    sum += parseInt(cnpj[i], 10) * multiplier;
+    multiplier = multiplier === 2 ? 9 : multiplier - 1;
+  }
+  const remainder2 = sum % 11;
+  const digit2 = remainder2 < 2 ? 0 : 11 - remainder2;
+
+  return parseInt(cnpj[12], 10) === digit1 && parseInt(cnpj[13], 10) === digit2;
+}
+
+export const TcuCertidoesSchema = z
+  .object({
+    cnpj: z
+      .string()
+      .describe("CNPJ number to query consolidated TCU certificates (with or without formatting)")
+      .transform((input) => input.replace(/\D/g, ""))
+      .refine((cnpj) => cnpj.length === 14 && /^\d{14}$/.test(cnpj), {
+        message: "CNPJ must be exactly 14 digits",
+      })
+      .refine(validateTcuCnpjChecksum, {
+        message: "Invalid CNPJ: checksum verification failed",
+      }),
+  })
+  .strict();

+  if (value === null || value === undefined) {
+    return null;
+  }
+  return sanitizeUntrustedText(
+    typeof value === "string" ? value : String(value),
+    maxChars,
+  );

+/**
+ * Query consolidated public integrity certificates for a CNPJ via TCU API.
+ */
+export async function executeTcuCertidoes(
+  args: TcuCertidoesArgs,
+  cache?: Cache,
+): Promise<LookupResult> {
+  const parsed = TcuCertidoesSchema.safeParse(args);
+  if (!parsed.success) {
+    const details = parsed.error.issues.map((issue) => issue.message).join("; ");
+    return { ok: false, error: details || "Invalid CNPJ" };
+  }
+
+  const cleanCnpj = parsed.data.cnpj;
+  const cacheKey = `tcu_certidoes:${cleanCnpj}`;
+
+  if (cache) {
+    const cached = await cache.get<Record<string, unknown>>(cacheKey);
+    if (cached && typeof cached === "object" && !Array.isArray(cached)) {
+      return {
+        ok: true,
+        data: attachLgpdNotice({
+          ...cached,
+          cached: true,
+        }),
+      };
+    }
+  }
+
+  const url = `${TCU_CERTIDOES_URL}/${cleanCnpj}?seEmitirPDF=false`;
+  const result = await httpJson(
+    url,
+    { Accept: "application/json" },
+    TCU_TIMEOUT_MS,
+  );
+
+  if (!result.ok) {
+    return {
+      ok: false,
+      error: `TCU Certidões API error: ${result.error || "request failed"}`,
+    };
+  }
+
+  const raw = result.data as Record<string, unknown> | null;
+  if (!raw || typeof raw !== "object" || Array.isArray(raw) || !Array.isArray(raw.certidoes)) {
+    return {
+      ok: false,
+      error: "TCU Certidões API error: unexpected response format",
+    };
+  }
+
+  const certidoes: TcuCertidaoItem[] = raw.certidoes
+    .filter(
+      (item): item is Record<string, unknown> =>
+        typeof item === "object" && item !== null && !Array.isArray(item),
+    )
+    .map((item) => ({
+      emissor: sanitizeRequiredText(item.emissor, 200),
+      tipo: sanitizeRequiredText(item.tipo, 200),
+      dataHoraEmissao: sanitizeRequiredText(item.dataHoraEmissao, 100),
+      descricao: sanitizeRequiredText(item.descricao, 1000),
+      situacao: sanitizeRequiredText(item.situacao, 100),
+      observacao: sanitizeOptionalText(item.observacao, 1000),
+      linkConsultaManual: sanitizeOptionalText(item.linkConsultaManual, 500),
+    }));
+
+  const has_positive_records = certidoes.some(
+    (item) =>
+      item.situacao !== "" &&
+      item.situacao !== "NADA_CONSTA" &&
+      item.situacao !== "SISTEMA_INDISPONIVEL",
+  );
+
+  const unavailable_issuers = certidoes
+    .filter((item) => item.situacao === "SISTEMA_INDISPONIVEL")
+    .map((item) => item.emissor)
+    .filter((emissor) => emissor.length > 0);
+
+  const razaoSocial =
+    typeof raw.razaoSocial === "string" && raw.razaoSocial.trim().length > 0
+      ? sanitizeUntrustedText(raw.razaoSocial, 300)
+      : undefined;
+
+  const payload = attachLgpdNotice({
+    cnpj: cleanCnpj,
+    ...(razaoSocial ? { razao_social: razaoSocial } : {}),
+    certidoes,
+    total_certidoes: certidoes.length,
+    has_positive_records,
+    unavailable_issuers,
+    cached: false,
+    source: "TCU — Consulta Consolidada de Pessoa Jurídica",
+    source_url: url,
+    fetchedAt: new Date().toISOString(),
+    untrusted_data_notice: UNTRUSTED_EXTERNAL_DATA_NOTICE,
+  });
+
+  if (cache) {
+    await cache.set(cacheKey, payload);
+  }
+
+  return { ok: true, data: payload };
+}
